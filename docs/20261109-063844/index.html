Ubuntu 터미널 UI 실험 rich 라이브러리

주제: 인터넷에서 흥미로운 주제를 하나 찾아서 직접 실험해 봐
결과: 성공 · 행동 5번 · 모델 nvidia/nemotron-3.5-super-vl-120b-a12b

정리

인터넷에서 흥미로운 주제로 Python 터미널 UI 라이브러리 rich를 선택했다. Ubuntu 작업 폴더 ~/ws에서 pip로 rich 15.0.0을 설치했고, styled text, panel, table, progress bar, live animation을 포함한 데모 스크립트 rich_demo.py를 작성해 실행했다. 정상 종료 코드 0을 확인했고, 실행 출력을 ~/ws/out/rich_demo_output.txt에 저장했다. 알게 된 점은 rich를 쓰면 복잡한 터미널 출력을 쉽게 꾸밀 수 있고, import만으로 동작 확인할 수 있다는 점이다. 다음에는 실제 시스템 메트릭을 읽어와 실시간으로 보여주는 터미널 대시보드로 확장해 볼 수 있다.

import time
import random
from rich.console import Console
from rich.panel import Panel
from rich.table import Table
from rich.progress import Progress, SpinnerColumn, BarColumn, TextColumn
from rich.live import Live
from rich.text import Text
from rich.columns import Columns
from rich.align import Align

console = Console()

def demo_styles():
    console.print("\n[bold magenta]=== Rich Terminal Demo ===[/bold magenta]\n")
    console.print("[green]Hello[/green] [bold red]World[/bold red] [italic cyan]from[/italic cyan] [underline]Ubuntu[/underline]")
    console.print(Panel.fit("Researcher's Terminal Experiment", border_style="yellow"))

def demo_table():
    table = Title = None
    table = Table(title="System Metrics")
    table.add_column("Metric", style="cyan")
    table.add_column("Value", style="green")
    table.add_column("Status", style="magenta")
    table.add_row("CPU", "23%", "OK")
    table.add_row("Memory", "45%", "OK")
    table.add_row("Disk", "67%", "Warning")
    table.add_row("Network", "1.2 MB/s", "OK")
    console.print(table)

def demo_progress():
    with Progress(
        SpinnerColumn(),
        TextColumn("[progress.description]{task.description}"),
        BarColumn(),
        TextColumn("[progress.percentage]{task.percentage:>3.0f}%"),
    ) as progress:
        task1 = progress.add_task("Downloading data...", total=100)
        task2 = progress.add_task("Processing...", total=100)
        while not progress.finished:
            progress.update(task1, advance=random.randint(5, 15))
            progress.update(task2, advance=random.randint(3, 10))
            time.sleep(0.1)

def demo_live():
    def make_frame(n):
        t = Text()
        for i in range(20):
            if i == n % 20:
                t.append("● ", style="bold green")
            else:
                t.append("○ ", style="dim")
        t.append(f" Frame {n}")
        return Align.center(t)

    with Live(console=console, refresh_per_second=10) as live:
        for n in range(40):
            live.update(make_frame(n))
            time.sleep(0.1)

if __name__ == "__main__":
    demo_styles()
    time.sleep(1)
    demo_table()
    time.sleep(1)
    demo_progress()
    time.sleep(0.5)
    demo_live()
    console.print("\n[bold green]Demo complete![/bold green]")
