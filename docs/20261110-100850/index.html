import plotext as plt
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt2
import os

def first_primes(n):
    res = []
    x = 2
    while len(res) < n:
        is_p = True
        for p in res:
            if p * p > x:
                break
            if x % p == 0:
                is_p = False
                break
        if is_p:
            res.append(x)
        x += 1
    return res

p = first_primes(100)
gaps = [p[i+1] - p[i] for i in range(len(p)-1)]

# Terminal visualization with plotext
plt.clf()
plt.title("First 100 Prime Numbers")
plt.plot(p, marker='dot')
plt.xlabel("Index")
plt.ylabel("Prime")
plt.show()

plt.clf()
plt.title("Prime Gaps")
plt.bar(gaps)
plt.xlabel("Index")
plt.ylabel("Gap")
plt.show()

# Save PNG with matplotlib
os.makedirs('out', exist_ok=True)
fig, axs = plt2.subplots(1, 2, figsize=(10, 4))
axs[0].plot(p, marker='o', markersize=3)
axs[0].set_title("First 100 Prime Numbers")
axs[0].set_xlabel("Index")
axs[0].set_ylabel("Prime")
axs[1].bar(range(len(gaps)), gaps)
axs[1].set_title("Prime Gaps")
axs[1].set_xlabel("Index")
axs[1].set_ylabel("Gap")
fig.tight_layout()
fig.savefig('out/primes.png', dpi=150)
print("Saved out/primes.png")
